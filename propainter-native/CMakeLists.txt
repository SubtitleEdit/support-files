cmake_minimum_required(VERSION 3.16)
project(propainter_native LANGUAGES C CXX)
set(GGML_SOURCE "" CACHE PATH "ggml source directory")
if(NOT EXISTS "${GGML_SOURCE}/include/ggml.h")
    message(FATAL_ERROR "Set GGML_SOURCE to the ggml source directory")
endif()
option(GGML_METAL "Build ggml Metal support" OFF)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${GGML_SOURCE}" ggml)
add_executable(propainter-checkpoint checkpoint.cpp)
target_compile_features(propainter-checkpoint PRIVATE cxx_std_17)
target_link_libraries(propainter-checkpoint PRIVATE ggml)
add_library(propainter-raft-correlation-core STATIC raft-correlation-core.cpp)
target_compile_features(propainter-raft-correlation-core PUBLIC cxx_std_17)
target_link_libraries(propainter-raft-correlation-core PUBLIC ggml)
add_executable(propainter-raft-correlation raft-correlation.cpp)
target_compile_features(propainter-raft-correlation PRIVATE cxx_std_17)
target_link_libraries(propainter-raft-correlation PRIVATE propainter-raft-correlation-core)
add_library(propainter-raft-core STATIC raft.cpp)
add_executable(propainter-raft raft-cli.cpp)
target_compile_features(propainter-raft PRIVATE cxx_std_17)
target_link_libraries(propainter-raft-core PUBLIC propainter-raft-correlation-core)
target_link_libraries(propainter-raft PRIVATE propainter-raft-core)
add_library(propainter-model-ops STATIC model-ops.cpp)
target_compile_features(propainter-model-ops PUBLIC cxx_std_17)
target_link_libraries(propainter-model-ops PUBLIC ggml)
target_link_libraries(propainter-raft-core PUBLIC propainter-model-ops)
add_library(propainter-flow-completion STATIC flow-completion.cpp)
target_link_libraries(propainter-flow-completion PUBLIC propainter-model-ops)
add_executable(propainter-complete-flow flow-completion-cli.cpp)
target_link_libraries(propainter-complete-flow PRIVATE propainter-flow-completion)
add_library(propainter-generator STATIC generator.cpp sparse-transformer.cpp)
target_link_libraries(propainter-generator PUBLIC propainter-model-ops)
add_executable(propainter-native propainter-cli.cpp renderer.cpp stream.cpp)
target_link_libraries(propainter-native PRIVATE propainter-raft-core propainter-flow-completion propainter-generator)
